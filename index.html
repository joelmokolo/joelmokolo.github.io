WITH trusted_data AS (
  SELECT
    source,
    COUNT(*) AS records,
    SUM(CASE WHEN is_valid
        THEN 1 ELSE 0 END)
        AS valid_records
  FROM raw_events
  GROUP BY source
)
SELECT *
FROM trusted_data
ORDER BY valid_records DESC;
